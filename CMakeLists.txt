cmake_minimum_required(VERSION 3.18)
project(iPTAM LANGUAGES CXX)
option(IPTAM_BUILD_TOOLS "Build calibrated headless replay" ON)
option(IPTAM_BUILD_TESTS "Build original-core regression tests" ON)
option(IPTAM_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(IPTAM_MOBILE_PAPER "Build experimental ISMAR2009-inspired mobile algorithms" OFF)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(IPTAM_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/xPTAMclean)
set(PTAM_SOURCES ATANCamera Bundle HomographyInit KeyFrame Map MapMaker MapPoint MiniPatch PatchFinder Relocaliser ShiTomasi SmallBlurryImage Tracker)
set(CORE_SOURCES)
foreach(name IN LISTS PTAM_SOURCES)
  list(APPEND CORE_SOURCES ${IPTAM_SOURCE_DIR}/iPTAM/${name}.cc)
endforeach()
if(IPTAM_MOBILE_PAPER)
  file(GLOB MOBILE_SOURCES CONFIGURE_DEPENDS ${IPTAM_SOURCE_DIR}/iPTAM/Mobile*.cc ${IPTAM_SOURCE_DIR}/iPTAM/RollingShutter.cc)
  list(APPEND CORE_SOURCES ${MOBILE_SOURCES})
endif()
file(GLOB CVD_SOURCES CONFIGURE_DEPENDS ${IPTAM_SOURCE_DIR}/iCVD/*.cc ${IPTAM_SOURCE_DIR}/iCVD/*.cxx)
list(APPEND CORE_SOURCES ${CVD_SOURCES} ${IPTAM_SOURCE_DIR}/iCVD/thread.cpp ${IPTAM_SOURCE_DIR}/iCVD/posix_memalign.cpp)
add_library(iptam_core STATIC ${CORE_SOURCES})
target_include_directories(iptam_core PUBLIC ${IPTAM_SOURCE_DIR}/iPTAM ${IPTAM_SOURCE_DIR}/iCVD ${IPTAM_SOURCE_DIR}/iTooN)
target_compile_definitions(iptam_core PUBLIC IPTAM_HEADLESS=1 CVD_HAVE_XMMINTRIN=0)
if(IPTAM_MOBILE_PAPER)
  target_compile_definitions(iptam_core PUBLIC IPTAM_MOBILE_PAPER=1)
endif()
target_compile_options(iptam_core PRIVATE -Wno-deprecated-declarations -Wno-unused-value)
if(APPLE)
  target_link_libraries(iptam_core PUBLIC "-framework Accelerate")
else()
  find_package(LAPACK REQUIRED)
  find_package(Threads REQUIRED)
  target_link_libraries(iptam_core PUBLIC ${LAPACK_LIBRARIES} Threads::Threads)
endif()
if(IPTAM_SANITIZE)
  target_compile_options(iptam_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(iptam_core PUBLIC -fsanitize=address,undefined)
endif()
if(IPTAM_BUILD_TOOLS)
  add_executable(iptam_replay tools/replay.cc)
  target_link_libraries(iptam_replay PRIVATE iptam_core)
endif()
if(IPTAM_BUILD_TESTS)
  enable_testing()
  if(IPTAM_MOBILE_PAPER)
    file(GLOB MOBILE_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/mobile_*_tests.cc)
    foreach(source IN LISTS MOBILE_TESTS)
      get_filename_component(test_name ${source} NAME_WE)
      add_executable(${test_name} ${source})
      target_link_libraries(${test_name} PRIVATE iptam_core)
      add_test(NAME ${test_name} COMMAND ${test_name})
      set_tests_properties(${test_name} PROPERTIES TIMEOUT 60)
    endforeach()
  endif()
  add_executable(iptam_core_tests tests/core_tests.cc)
  target_link_libraries(iptam_core_tests PRIVATE iptam_core)
  add_test(NAME core_regressions COMMAND iptam_core_tests)
  set_tests_properties(core_regressions PROPERTIES TIMEOUT 120)
  add_executable(iptam_patchfinder_tests tests/patchfinder_tests.cc)
  target_link_libraries(iptam_patchfinder_tests PRIVATE iptam_core)
  add_test(NAME patchfinder_regressions COMMAND iptam_patchfinder_tests)
  set_tests_properties(patchfinder_regressions PROPERTIES TIMEOUT 30)
  add_executable(iptam_bundle_solver_tests tests/bundle_solver_tests.cc)
  target_link_libraries(iptam_bundle_solver_tests PRIVATE iptam_core)
  target_compile_definitions(iptam_bundle_solver_tests PRIVATE IPTAM_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME bundle_solver_regressions COMMAND iptam_bundle_solver_tests)
  set_tests_properties(bundle_solver_regressions PROPERTIES TIMEOUT 5)
  add_executable(iptam_equidistant_camera_tests tests/equidistant_camera_tests.cc)
  target_link_libraries(iptam_equidistant_camera_tests PRIVATE iptam_core)
  add_test(NAME equidistant_camera_regressions COMMAND iptam_equidistant_camera_tests)
  set_tests_properties(equidistant_camera_regressions PROPERTIES TIMEOUT 30)
  if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJCXX)
    add_executable(iptam_manifest_tests tests/replay_manifest_tests.mm)
    target_compile_options(iptam_manifest_tests PRIVATE -fobjc-arc)
    target_link_libraries(iptam_manifest_tests PRIVATE "-framework Foundation")
    add_test(NAME replay_manifest_regressions COMMAND iptam_manifest_tests)
    if(IPTAM_MOBILE_PAPER)
      add_executable(mobile_manifest_tests tests/mobile_manifest_tests.mm)
      target_compile_options(mobile_manifest_tests PRIVATE -fobjc-arc)
      target_link_libraries(mobile_manifest_tests PRIVATE iptam_core "-framework Foundation")
      add_test(NAME mobile_manifest_tests COMMAND mobile_manifest_tests)
    endif()
  endif()
endif()
